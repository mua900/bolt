# Example
# paths are relative to an asset/ folder in the base path (near the executable)

# syntax
# <kind> <scope optional> <name> <path> [attributes] [optional] [lazy]
# if you omit scope it is inferred as file
# attributes are in key value format

# font
font file AwesomeFont font/AwesomeFont-Regular.ttf size=32

# image
image file MyImage images/image.png mip_levels=3
image file MyVectorImage images/image.svg

# folder
# gets all the files in the folder and uses their file names for their asset names
# this doesn't recurse into subfolders
image folder Icons icon

# audio uses sdl mixer to load and playback
# any format supported by sdl mixer is supported by this library
audio file AwesomeTrack audio/music/track.mp3

# shader
# the extension can be spv, dxil or msl and only relevant ones for the platforms will be loaded
# if you want details search for 'loadShader' in the source code
shader file VertexShader shader/binary/vertex.spv       stage=vertex   nUniforms=1
shader file FragmentShader shader/binary/fragment.spv   stage=fragment
shader file VertexShader shader/binary/vertex.dxil      stage=vertex   nUniforms=1
shader file FragmentShader shader/binary/fragment.dxil  stage=fragment
shader file VertexShader shader/binary/vertex.msl       stage=vertex   nUniforms=1
shader file FragmentShader shader/binary/fragment.msl   stage=fragment
